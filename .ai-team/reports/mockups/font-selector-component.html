Font Selector: Before & After

The Countdown Timer and Goal Bar pages use 24+ radio buttons to select fonts. This creates an overwhelming UI that's hard to scan. A dropdown select with live preview is much more user-friendly.

// FontSelectorComponent.razor - Reusable component for font selection <MudStack Spacing="2"> <MudSelect T="FontFamily" Label="Font" Variant="Variant.Filled" @bind-Value="SelectedFont" ValueChanged="OnFontChanged"> @foreach (var font in FontOptions) { <MudSelectItem Value="@font">@font.DisplayName</MudSelectItem> } </MudSelect> <MudPaper Class="pa-4 text-center gifbot-form-panel" Elevation="0"> <MudText Typo="Typo.caption" Class="text-white-50">Preview</MudText> <div style="font-family: @SelectedFont.CssName; font-size: 32px; margin-top: 8px;"> @PreviewText </div> </MudPaper> </MudStack> @code { [Parameter] public FontFamily SelectedFont { get; set; } [Parameter] public EventCallback<FontFamily> SelectedFontChanged { get; set; } [Parameter] public string PreviewText { get; set; } = "12:34:56"; private List<FontFamily> FontOptions = new() { new("Arial", "Arial, sans-serif"), new("Comic Sans", "'Comic Sans MS', cursive"), new("Impact", "Impact, sans-serif"), // ... etc }; private async Task OnFontChanged(FontFamily font) { SelectedFont = font; await SelectedFontChanged.InvokeAsync(font); } }
